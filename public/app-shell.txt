function zClamp(){
  // stranica ostaje gdje je korisnik ostavi; rub može doći najviše do sredine ekrana
  const vw = zview.clientWidth, vh = zview.clientHeight, w = ZW * zcount * zs, h = ZH * zs, mx = vw / 2, my = vh / 2;
  zx = Math.min(Math.max(mx, vw - w), Math.max(Math.min(mx - w, 0), zx));
  zy = Math.min(Math.max(my, vh - h), Math.max(Math.min(my - h, 0), zy));
}

  const vw = zview.clientWidth || innerWidth, vh = zview.clientHeight || innerHeight - 70, m = vw < 640 ? 12 : 32;
  zfit = Math.min((vw - m) / (ZW * zcount), (vh - m) / ZH);

function zAt(px, py, f, anim){
  const ns = Math.max(zfit, Math.min(zfit * 8, zs * f));
  zx = px - (px - zx) * ns / zs; zy = py - (py - zy) * ns / zs; zs = ns;

/* ---------- routing ----------
   (prazno)/#brosure → pregled svih brošura
   #lamela-a        → panel s odabranom brošurom
   #z-stanovi / #z-garaze / #z-poslovni → klijentski paket brošura
   #k-lamela-a      → klijentski link: prezentacija brošure
   #k-lamela-a-12   → klijentski link: prezentacija + zum na stranicu 12 */
function route(){
  const h = location.hash.slice(1);
  let pid = null;
  try { pid = new URLSearchParams(location.search).get('p'); } catch(e) {}
  const z = h.match(/^z-([a-z0-9-]+)$/);
  if (z) pid = z[1];
  if (pid){
    const pk = PACKS.find(x => x.id === pid);
    if (pk) return openPack(pk);
    if (CLIENT_ONLY) return clientMessage('Ponuda nije pronađena', 'Link koji ste otvorili nije ispravan. Kontaktirajte naš prodajni tim putem dugmeta Kontakt.');
  }
  let kq = null; try { kq = new URLSearchParams(location.search).get('k'); } catch(e) {}
  const k = (kq ? 'k-' + kq : h).match(/^k-([a-z-]+?)(?:-(\d+))?$/);
  if (k){
    const b = BROS.find(x => x.id === k[1]);
    if (b){
      selectBro(b.id, true);
      document.body.classList.add('client');
      if (CLIENT_ONLY) document.body.classList.add('cview');
      const i = k[2] ? Math.max(0, Math.min(b.pages - 1, +k[2] - 1)) : 0;
      present(b, i);
      if (k[2]) openPageZoom(b, i);
      return;
    }
  }
  if (CLIENT_ONLY && CFG.loginUrl){ location.replace(CFG.loginUrl); return; }
  if (CLIENT_ONLY) return clientMessage('Sarajka brošure', 'Za pregled ponude otvorite link ili skenirajte QR kod koji ste dobili od našeg prodajnog tima.');
  if (BROS.some(b => b.id === h)) selectBro(h, true); else showHome(true);
}
